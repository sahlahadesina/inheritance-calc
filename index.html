Inheritance Calculator

Enter the total estate value, select the gender of the deceased,
and indicate the surviving relatives to calculate shares.

        from pyscript import document
        from engine import Engine

        def calculate_ui(event):
            try:
                estate_val_str = document.querySelector("#e_estate").value
                if not estate_val_str:
                    estate_val_str = "0"
                estate_val = float(estate_val_str)
            except ValueError:
                document.querySelector("#output").innerText = "Input Error: Please enter a valid number for the Estate."
                return

            engine = Engine(estate_val)
            gender = document.querySelector('input[name="gender"]:checked').value

            if gender == "Male":
                wives_val = document.querySelector("#e_wives").value
                engine.add("wife", int(wives_val) if wives_val else 0)
            else:
                husband_checked = document.querySelector("#husband_var").checked
                engine.add("husband", 1 if husband_checked else 0)

            # Helper functions for UI element retrieval
            def get_spin(selector):
                val = document.querySelector(selector).value
                return int(val) if val else 0
                
            def get_check(selector):
                return 1 if document.querySelector(selector).checked else 0

            # Add all heirs linked to the HTML IDs
            engine.add("son", get_spin("#e_sons"))
            engine.add("daughter", get_spin("#e_daughters"))
            
            engine.add("father", get_check("#father_var"))
            engine.add("mother", get_check("#mother_var"))
            engine.add("paternal_grandfather", get_check("#pat_grandpa_var"))
            engine.add("maternal_grandfather", get_check("#mat_grandpa_var"))
            engine.add("paternal_grandmother", get_check("#pat_grandma_var"))
            engine.add("maternal_grandmother", get_check("#mat_grandma_var"))
            
            engine.add("brother", get_spin("#e_bros"))
            engine.add("sister", get_spin("#e_siss"))
            engine.add("paternal_brother", get_spin("#e_pat_bros"))
            engine.add("paternal_sister", get_spin("#e_pat_siss"))
            engine.add("maternal_brother", get_spin("#e_mat_bros"))
            engine.add("maternal_sister", get_spin("#e_mat_siss"))
            
            engine.add("paternal_uncle", get_spin("#e_pat_uncle"))
            engine.add("maternal_uncle", get_spin("#e_mat_uncle"))

            result = engine.calc()

            # Display results
            output = document.querySelector("#output")
            output.innerText = ""
            
            if not result:
                output.innerText = "No specific fixed shares calculated based on current inputs."
                return

            output.innerText += f"Total Estate Value: {estate_val:,.2f}\n"
            output.innerText += "-" * 40 + "\n"
            
            for k, v in result.items():
                count = engine.heirs.get(k, 1)
                calculated_value = float(estate_val) * float(v)
                display_name = k.replace('_', ' ').title()

                if count > 1:
                    per_person_frac = v / count
                    per_person_val = calculated_value / count
                    output.innerText += f"{display_name}s ({count}): Total {v} (Value: {calculated_value:,.4g})\n"
                    output.innerText += f"  └─ Each gets: {per_person_frac} (Value: {per_person_val:,.4g})\n"
                else:
                    output.innerText += f"{display_name}: {v}  (Value: {calculated_value:,.4g})\n"
